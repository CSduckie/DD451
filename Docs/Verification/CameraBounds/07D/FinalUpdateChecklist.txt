$baseline = Get-Content -LiteralPath 'Docs/Verification/TriggerInteraction/07C/SerializedAssetAudit.json' -Raw | ConvertFrom-Json
$manifest = Get-Content -LiteralPath 'Docs/Verification/TriggerInteraction/07C/AssetBackupManifest.json' -Raw | ConvertFrom-Json
$rows = @(foreach ($entry in $manifest) {
    $known = @($baseline.files | Where-Object path -eq $entry.path)
    if ($known.Count -ne 1) { throw "Missing or duplicate semantic baseline: $($entry.path)" }
    $current = (Get-FileHash -LiteralPath $entry.path -Algorithm SHA256).Hash
    $backup = (Get-FileHash -LiteralPath $entry.backup -Algorithm SHA256).Hash
    [pscustomobject]@{
        path = $entry.path
        backupValid = ($backup -eq $entry.sha256)
        changedFromBefore = ($current -ne $entry.sha256)
        changedSinceSemanticAudit = ($current -ne $known[0].currentSha256)
        currentHash = $current
    }
})
[pscustomobject]@{
    utc = [DateTime]::UtcNow.ToString('o')
    files = $rows.Count
    invalidBackups = @($rows | Where-Object { -not $_.backupValid })
    changedFromBefore = @($rows | Where-Object changedFromBefore)
    changedSinceSemanticAudit = @($rows | Where-Object changedSinceSemanticAudit)
} | ConvertTo-Json -Depth 5
