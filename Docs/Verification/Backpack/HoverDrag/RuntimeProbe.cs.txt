    // 动态编译的嵌套DTO不用JsonUtility；只读公有实例字段，避免Vector.normalized等属性递归。
    private sealed class PublicInstanceFieldsResolver : DefaultContractResolver
    {
        protected override IList<JsonProperty> CreateProperties(Type type, MemberSerialization memberSerialization)
        {
            List<JsonProperty> properties = new List<JsonProperty>();
            foreach (FieldInfo field in type.GetFields(BindingFlags.Public | BindingFlags.Instance).OrderBy(f => f.Name, StringComparer.Ordinal))
            {
                JsonProperty property = base.CreateProperty(field, MemberSerialization.Fields);
                property.PropertyName = field.Name;
                property.Readable = true;
                property.Writable = !field.IsInitOnly;
                property.Ignored = false;
                properties.Add(property);
            }
            return properties;
        }
    }

    public static IEnumerator Run(LootTestDriver driver)
    {
        if (Running) { yield break; }
        Running = true;
        Completed = false;
        Error = string.Empty;
        cancelRequested = false;
        Context context = new Context();
        context.driver = driver;
        context.startedUtc = UtcNow();
        OutputDirectory = Path.Combine(EvidenceRoot, DateTime.UtcNow.ToString("yyyyMMdd-HHmmss-fff", CultureInfo.InvariantCulture));
        Stack<IEnumerator> stack = new Stack<IEnumerator>();
        try
        {
            // execute_code同步入口先保存/临时启用后台运行，确保Game失焦时首个yield仍能推进。
            context.originalBackground = Application.runInBackground;
            context.backgroundCaptured = true;
            Application.runInBackground = true;
            // StartCoroutine可从Editor execute_code同步进入；先等真实Play帧再读取Screen和输入状态。
            yield return null;
            // 手动展开有限的子协程，任何阶段异常都回到本finally，避免Unity吞掉子协程异常而遗漏恢复。
            stack.Push(Sequence(context));
            while (stack.Count > 0)
            {
                if (cancelRequested) { Error = "Cancelled by caller."; break; }
                object yielded = null;
                bool advanced = false;
                bool failed = false;
                try
                {
                    advanced = stack.Peek().MoveNext();
                    if (advanced) { yielded = stack.Peek().Current; }
                }
                catch (Exception exception)
                {
                    Error = exception.ToString();
                    failed = true;
                }
                if (failed) { break; }
                if (!advanced)
                {
                    IDisposable finished = stack.Pop() as IDisposable;
                    if (finished != null) { finished.Dispose(); }
                    continue;
                }
                IEnumerator nested = yielded as IEnumerator;
                if (nested != null) { stack.Push(nested); }
                else { yield return yielded; }
            }
            Completed = stack.Count == 0 && string.IsNullOrEmpty(Error);
        }
        finally
        {
            while (stack.Count > 0)
            {
                IDisposable unfinished = stack.Pop() as IDisposable;
                if (unfinished == null) { continue; }
                try { unfinished.Dispose(); }
                catch (Exception exception) { AppendError("Enumerator cleanup", exception); }
            }
            Cleanup(context);
            Running = false;
            if (!string.IsNullOrEmpty(Error)) { Completed = false; }
            try
            {
                Directory.CreateDirectory(OutputDirectory);
                Snapshot after = context.driver != null && context.driver.IsReady ? ReadSnapshot(context, "FinalAfterCleanup") : null;
                if (after != null) { WriteJson("FinalAfterCleanup.json", after); }
                RunSummary summary = new RunSummary();
                summary.startedUtc = context.startedUtc;
                summary.finishedUtc = UtcNow();
                summary.completed = Completed;
                summary.error = Error;
                summary.evidenceDirectory = OutputDirectory;
                summary.inputEvidence = "Synthetic QueueStateEvent -> InputAction/EventSystem/UGUI; not native mouse acceptance.";
                summary.apiAssistance = IncludeLootDamage
                    ? "Driver context changes, RelocateWithinRun, reach-validated TryOpenSource. No direct drag Begin/Update/End, transfer, reveal or round-reset API."
                    : "Driver context changes and initial language selection. No direct drag Begin/Update/End, transfer, reveal or round-reset API.";
                summary.lootDamageIncluded = IncludeLootDamage;
                summary.healthRestored = false;
                summary.originalBackground = context.originalBackground;
                summary.finalBackground = Application.runInBackground;
                summary.note = (IncludeLootDamage ? "Real nonlethal damage remains applied. " : "Default ten-stage sequence causes no damage. ")
                    + "No authoritative item rollback is performed on success or failure. Sessions are closed; original context, position, language and input/background settings are restored.";
                summary.transferCompletions = context.transfers;
                summary.captures = context.captures.ToArray();
                summary.checks = context.checks.ToArray();
                summary.observedUnityErrors = context.logs.ToArray();
                summary.before = context.before;
                summary.after = after;
                WriteJson("MotionSamples.json", new MotionList { samples = context.motion.ToArray() });
                WriteJson("Summary.json", summary);
                File.WriteAllText(Path.Combine(EvidenceRoot, "LatestRun.json"), Serialize(new LatestRun
                { directory = OutputDirectory, completed = Completed, error = Error }, true));
            }
            catch (Exception exception)
            {
                AppendError("Summary write", exception);
                Completed = false;
            }
        }
    }

    private static IEnumerator Pointer(Context context, Vector2 position, bool left)
    {
        InputSystem.QueueStateEvent(context.mouse, new MouseState { position = position, buttons = (ushort)(left ? 1 : 0) });
        yield return null;
        yield return null;
        context.motion.Add(new MotionData { frame = Time.frameCount, realtime = Time.realtimeSinceStartup,
            pointer = position, held = left, images = ReadImages() });
    }

    private static IEnumerator KeyPress(Context context, Key key)
    {
        InputSystem.QueueStateEvent(context.keyboard, new KeyboardState(key));
        yield return null;
        yield return null;
        InputSystem.QueueStateEvent(context.keyboard, new KeyboardState());
        yield return null;
        yield return null;
    }

    private static IEnumerator StartDrag(Context context, Vector2 point)
    {
        yield return Pointer(context, point, false);
        yield return Pointer(context, point, true);
        yield return Pointer(context, point + Vector2.right * Mathf.Max(20f, EventSystem.current.pixelDragThreshold + 4f), true);
    }

    private static IEnumerator Capture(Context context, string name)
    {
        yield return new WaitForEndOfFrame();
        Texture2D texture = null;
        try
        {
            Snapshot snapshot = ReadSnapshot(context, name);
            snapshot.imageFile = name + ".png";
            texture = ScreenCapture.CaptureScreenshotAsTexture();
            Require(texture != null && texture.width == 1920 && texture.height == 1080, "Screenshot must contain the actual 1920x1080 Game frame.");
            File.WriteAllBytes(Path.Combine(OutputDirectory, snapshot.imageFile), texture.EncodeToPNG());
            WriteJson(name + ".json", snapshot);
            context.captures.Add(name);
        }
        finally { if (texture != null) { Object.Destroy(texture); } }
    }

    private static void CheckGhost(Context context, Vector2 pointer, Vector2 grab, Vector2 size)
    {
        BackpackDragImage image = context.driver.backpackUI.view.DragImage;
        Require(image != null && image.IsVisible && image.SourceRect == context.trackedSource, "Expected live shared image and original event receiver.");
        Rect rect = ScreenRectangle(image.RectTransform);
        Require(Mathf.Abs(rect.xMin - (pointer.x - grab.x)) < 0.6f && Mathf.Abs(rect.yMax - (pointer.y + grab.y)) < 0.6f, "Ghost pixel grab offset jumped.");
        Require(Mathf.Abs(rect.width - size.x) < 0.2f && Mathf.Abs(rect.height - size.y) < 0.2f, "Ghost dimensions differ from the item orientation.");
        Require(Object.FindObjectsByType<BackpackDragImage>(FindObjectsSortMode.None).Count(i => i.IsVisible) == 1, "Expected exactly one visible drag image.");
        Require(image.GetComponentsInChildren<Graphic>(true).All(g => !g.raycastTarget), "A ghost Graphic intercepts raycasts.");
        List<RaycastResult> hits = new List<RaycastResult>();
        EventSystem.current.RaycastAll(new PointerEventData(EventSystem.current) { position = pointer }, hits);
        Require(!hits.Any(h => h.gameObject.transform == image.transform || h.gameObject.transform.IsChildOf(image.transform)), "Ghost appears in the actual UI raycast results.");
        Require(context.trackedSource.gameObject.activeInHierarchy && SourceAlpha(context.trackedSource) < 1f, "Drag disabled or failed to dim its original event receiver.");
    }

    private static Vector2 GrabPoint(RectTransform source, float x, float y)
    {
        Require(source != null, "Required item RectTransform is missing.");
        Rect rect = ScreenRectangle(source);
        return new Vector2(rect.xMin + x, rect.yMax - y);
    }

    private static Vector2 BackpackPoint(LootTestDriver driver, int x, int y)
    {
        BackpackView view = driver.backpackUI.view;
        return CellPoint(view.gridRoot, view.padding, view.cellSize, view.gap, x, y);
    }

    private static Vector2 WarehousePoint(LootTestDriver driver, int x, int y)
    {
        WarehouseView view = driver.warehouseUI.view;
        return CellPoint(view.gridRoot, view.padding, view.EffectiveCellSize, view.gap, x, y);
    }

    private static Vector2 CellPoint(RectTransform root, float padding, float size, float gap, int x, int y)
    {
        Canvas.ForceUpdateCanvases();
        return RectTransformUtility.WorldToScreenPoint(null, root.TransformPoint(new Vector3(
            root.rect.xMin + padding + x * (size + gap) + size * 0.35f,
            root.rect.yMax - padding - y * (size + gap) - size * 0.35f, 0f)));
    }

    private static Vector2 BackpackGap(LootTestDriver driver, int x, int y)
    {
        BackpackView view = driver.backpackUI.view;
        return RectTransformUtility.WorldToScreenPoint(null, view.gridRoot.TransformPoint(new Vector3(
            view.gridRoot.rect.xMin + view.padding + x * (view.cellSize + view.gap) + view.cellSize + view.gap * 0.5f,
            view.gridRoot.rect.yMax - view.padding - y * (view.cellSize + view.gap) - 25f, 0f)));
    }

    private static bool VisibleGhost(LootTestDriver driver)
    {
        return driver.backpackUI.view.DragImage != null && driver.backpackUI.view.DragImage.IsVisible;
    }

    private static float SourceAlpha(RectTransform source)
    {
        if (source == null) { return -1f; }
        CanvasGroup group = source.GetComponent<CanvasGroup>();
        return group != null ? group.alpha : 1f;
    }

    private static Rect ScreenRectangle(RectTransform rect)
    {
        if (rect == null) { return Rect.zero; }
        Vector3[] corners = new Vector3[4];
        rect.GetWorldCorners(corners);
        Canvas canvas = rect.GetComponentInParent<Canvas>();
        Camera camera = canvas != null && canvas.rootCanvas.renderMode != RenderMode.ScreenSpaceOverlay ? canvas.rootCanvas.worldCamera : null;
        Vector2 min = RectTransformUtility.WorldToScreenPoint(camera, corners[0]);
        Vector2 max = RectTransformUtility.WorldToScreenPoint(camera, corners[2]);
        return new Rect(min, max - min);
    }

    private static void Check(Context context, bool condition, string message)
    {
        context.checks.Add((condition ? "PASS " : "FAIL ") + message);
        Require(condition, message);
    }

    private static void Require(bool condition, string message)
    {
        if (!condition) { throw new InvalidOperationException(message); }
    }

    private static void Cleanup(Context context)
    {
        // 每一项单独保护，某个已卸载对象失效也不能阻止恢复真实输入。
        Attempt("Close loot", delegate { if (context.driver != null && context.driver.lootUI != null) { context.driver.lootUI.CloseSource(); } });
        Attempt("Close backpack", delegate { if (context.driver != null && context.driver.backpackUI != null) { context.driver.backpackUI.Close(); } });
        Attempt("Close warehouse", delegate { if (context.driver != null && context.driver.warehouseUI != null) { context.driver.warehouseUI.Close(); } });
        if (context.settingsCaptured && context.driver != null && context.driver.IsReady)
        {
            Attempt("Restore driver context", delegate
            {
                if (context.originalShelter) { context.driver.EnterShelter(); } else { context.driver.EnterExploration(); }
                context.driver.backpackUI.Close();
                context.driver.warehouseUI.Close();
                context.driver.lootUI.CloseSource();
            });
            Attempt("Restore player position", delegate { if (context.driver.player != null && !context.driver.player.IsDead) { context.driver.player.RelocateWithinRun(context.originalPlayerPosition); } });
            Attempt("Restore language", delegate { if (LocalizationManager.Instance != null) { LocalizationManager.Instance.SetLanguage(context.originalLanguage); } });
        }
        Attempt("Remove keyboard", delegate { if (context.keyboard != null && context.keyboard.added) { InputSystem.RemoveDevice(context.keyboard); } });
        Attempt("Remove mouse", delegate { if (context.mouse != null && context.mouse.added) { InputSystem.RemoveDevice(context.mouse); } });
        if (context.settingsCaptured)
        {
            Attempt("Restore InputSettings", delegate { InputSystem.settings = context.originalSettings; });
        }
        if (context.backgroundCaptured)
        {
            Attempt("Restore background", delegate { Application.runInBackground = context.originalBackground; });
        }
        Attempt("Destroy temporary InputSettings", delegate { if (context.temporarySettings != null) { Object.Destroy(context.temporarySettings); } });
        foreach (InputDevice device in context.disabledDevices)
        {
            InputDevice restored = device;
            Attempt("Restore device " + restored.name, delegate { if (restored.added && !restored.enabled) { InputSystem.EnableDevice(restored); } });
        }
        Attempt("Unsubscribe completion", delegate { if (context.driver != null && context.driver.coordinator != null && context.onCompleted != null) { context.driver.coordinator.Completed -= context.onCompleted; } });
        Attempt("Unsubscribe logs", delegate { if (context.onLog != null) { Application.logMessageReceived -= context.onLog; } });
    }

    private static void Attempt(string label, Action action)
    {
        try { action(); }
        catch (Exception exception) { AppendError(label, exception); }
    }

    private static void AppendError(string label, Exception exception)
    {
        Error = (string.IsNullOrEmpty(Error) ? string.Empty : Error + "\n") + label + ": " + exception;
    }

    private static string UtcNow() { return DateTime.UtcNow.ToString("o", CultureInfo.InvariantCulture); }

    {
        return JsonConvert.SerializeObject(value, pretty ? Formatting.Indented : Formatting.None, JsonSettings);
    }
    private static void WriteJson(string name, object value) { File.WriteAllText(Path.Combine(OutputDirectory, name), Serialize(value, true)); }

    private static Snapshot ReadSnapshot(Context context, string stage)
    {
        LootTestDriver driver = context.driver;
        Snapshot snapshot = new Snapshot();
        snapshot.stage = stage;
        snapshot.utc = UtcNow();
        snapshot.frame = Time.frameCount;
        snapshot.realtime = Time.realtimeSinceStartup;
        snapshot.gameTime = Time.time;
        snapshot.timeScale = Time.timeScale;
        snapshot.screenWidth = Screen.width;
        snapshot.screenHeight = Screen.height;
        snapshot.activeScene = SceneManager.GetActiveScene().path;
        snapshot.loadedScenes = Enumerable.Range(0, SceneManager.sceneCount).Select(i => SceneManager.GetSceneAt(i).path).ToArray();
        snapshot.language = LocalizationManager.Instance != null ? LocalizationManager.Instance.CurrentLanguage.ToString() : "unavailable";
        snapshot.health = driver.player.health.CurrentHealth;
        snapshot.maxHealth = driver.player.health.MaxHealth;
        snapshot.dead = driver.player.IsDead;
        snapshot.playerPosition = driver.player.Position;
        snapshot.shelter = driver.IsShelter;
        snapshot.ev = driver.shelter.CurrentEv;
        snapshot.backpackOpen = driver.backpackUI.IsOpen;
        snapshot.warehouseOpen = driver.warehouseUI.IsOpen;
        snapshot.lootOpen = driver.lootUI.IsOpen;
        snapshot.backpackToken = driver.backpackUI.DragToken;
        snapshot.warehouseToken = driver.warehouseUI.DragToken;
        snapshot.lootSession = driver.lootUI.SessionToken;
        snapshot.backpackSelectedId = driver.backpackUI.SelectedInstanceId;
        snapshot.backpackPreview = driver.backpackUI.PreviewResult.ToString();
        snapshot.previewX = driver.backpackUI.PreviewX;
        snapshot.previewY = driver.backpackUI.PreviewY;
        snapshot.previewRotated = driver.backpackUI.PreviewRotated;
        Vector2Int grabOffset = driver.backpackUI.DragGrabOffset;
        snapshot.grabOffset = new Int2Data { x = grabOffset.x, y = grabOffset.y };
        snapshot.feedback = driver.backpackUI.FeedbackKey;
        snapshot.pointer = context.mouse != null && context.mouse.added ? context.mouse.position.ReadValue() : Vector2.zero;
        snapshot.leftHeld = context.mouse != null && context.mouse.added && context.mouse.leftButton.isPressed;
        snapshot.transfers = context.transfers;
        snapshot.background = Application.runInBackground;
        snapshot.inputDevices = InputSystem.devices.Select(d => d.deviceId + ":" + d.name + ":" + d.enabled).ToArray();
        snapshot.inventory = ReadInventory(driver);
        snapshot.images = ReadImages();
        snapshot.trackedSourceRect = RectData.From(ScreenRectangle(context.trackedSource));
        snapshot.trackedSourceAlpha = SourceAlpha(context.trackedSource);
        snapshot.trackedSourceActive = context.trackedSource != null && context.trackedSource.gameObject.activeInHierarchy;
        snapshot.sources = ReadSources(driver);
        snapshot.itemViews = ReadItemViews(driver);
        return snapshot;
    }

    private static InventoryData ReadInventory(LootTestDriver driver)
    {
        InventoryData data = new InventoryData();
        data.round = driver.items.CurrentRound;
        data.instanceCount = driver.items.InstanceCount;
        ItemInstance[] instances = driver.items.GetInstances().ToArray();
        data.distinctInstanceCount = instances.Select(i => i.InstanceId).Distinct(StringComparer.Ordinal).Count();
        Require(instances.Length == data.instanceCount && data.distinctInstanceCount == data.instanceCount
            && instances.All(i => !string.IsNullOrEmpty(i.InstanceId)),
            "ItemManager instances must have nonempty distinct InstanceIds and exactly InstanceCount identities.");
        data.items = instances.OrderBy(i => i.InstanceId, StringComparer.Ordinal).Select(i => new ItemData
        {
            runtimeObjectIdentity = RuntimeHelpers.GetHashCode(i),
            instanceId = i.InstanceId, definitionId = i.DefinitionId, uniqueId = i.UniqueId, owner = i.Owner.ToString(),
            containerId = i.ContainerId, sourceId = i.SourceId, originalSpawnPointId = i.OriginalSpawnPointId,
            useRound = i.UseRound, usesThisRound = i.UsesThisRound, retained = i.IsRetained,
            pendingRecovery = i.IsPendingRecovery, recoveryRound = i.RecoveryRound,
            lastLossReason = i.LastLossReason.HasValue ? i.LastLossReason.Value.ToString() : string.Empty,
            spawnClaimed = i.SpawnClaimed, ev = i.Definition != null ? i.Definition.ev : -1,
            definitionObjectId = i.Definition != null ? i.Definition.GetInstanceID() : 0,
            definitionWidth = i.Definition != null ? i.Definition.width : 0,
            definitionHeight = i.Definition != null ? i.Definition.height : 0
        }).ToArray();
        data.backpack = driver.backpack.GetPlacements().OrderBy(p => p.InstanceId).Select(p => new PlacementData
        { instanceId = p.InstanceId, x = p.X, y = p.Y, width = p.Width, height = p.Height, rotated = p.Rotated }).ToArray();
        data.warehouse = driver.warehouse.GetPlacements().OrderBy(p => p.InstanceId).Select(p => new PlacementData
        { instanceId = p.InstanceId, x = p.X, y = p.Y, width = p.Width, height = p.Height, rotated = p.Rotated }).ToArray();
        return data;
    }

    private static ImageData[] ReadImages()
    {
        return Object.FindObjectsByType<BackpackDragImage>(FindObjectsInactive.Include, FindObjectsSortMode.None).Select(image => new ImageData
        {
            objectId = image.GetInstanceID(), visible = image.IsVisible, token = image.Token,
            owner = image.Owner != null ? image.Owner.GetType().Name + ":" + image.Owner.GetInstanceID() : string.Empty,
            imageRect = RectData.From(ScreenRectangle(image.RectTransform)), sourceRect = RectData.From(ScreenRectangle(image.SourceRect)),
            sourceObjectId = image.SourceRect != null ? image.SourceRect.GetInstanceID() : 0,
            sourceActive = image.SourceRect != null && image.SourceRect.gameObject.activeInHierarchy, sourceAlpha = SourceAlpha(image.SourceRect),
            text = string.Join("\n", image.GetComponentsInChildren<TMPro.TMP_Text>(true).Select(t => t.text).ToArray()),
            graphics = image.GetComponentsInChildren<Graphic>(true).Select(g => new GraphicData
            { name = g.name, active = g.isActiveAndEnabled, raycast = g.raycastTarget, culled = g.canvasRenderer.cull,
                inheritedAlpha = g.canvasRenderer.GetInheritedAlpha(), color = g.color }).ToArray()
        }).ToArray();
    }

    private static SourceData[] ReadSources(LootTestDriver driver)
    {
        Dictionary<string, LootSourceState> sources = new Dictionary<string, LootSourceState>();
        foreach (LootBox box in Object.FindObjectsByType<LootBox>(FindObjectsInactive.Include, FindObjectsSortMode.None))
        { if (box.State != null) { sources[box.State.SourceId] = box.State; } }
        foreach (GroundItem ground in Object.FindObjectsByType<GroundItem>(FindObjectsInactive.Include, FindObjectsSortMode.None))
        { if (ground.State != null) { sources[ground.State.SourceId] = ground.State; } }
        if (driver.lootUI.Source != null) { sources[driver.lootUI.Source.SourceId] = driver.lootUI.Source; }
        return sources.Values.OrderBy(s => s.SourceId).Select(s => new SourceData
        {
            sourceId = s.SourceId, mapId = s.MapId, round = s.Round, ground = s.IsGround, generated = s.Generated,
            width = s.GridWidth, height = s.GridHeight, searchEntryId = s.SearchEntryId, searchElapsed = s.SearchElapsed,
            entries = s.Entries.Select(e => new EntryData { entryId = e.EntryId, instanceId = e.InstanceId,
                revealed = e.Revealed, taken = e.Taken, hasLayout = e.HasLayout, x = e.LayoutX, y = e.LayoutY,
                width = e.LayoutWidth, height = e.LayoutHeight, rotated = e.LayoutRotated }).ToArray()
        }).ToArray();
    }

    private static ItemViewData[] ReadItemViews(LootTestDriver driver)
    {
        List<ItemViewData> rows = new List<ItemViewData>();
        foreach (BackpackItemView view in driver.backpackUI.view.itemRoot.GetComponentsInChildren<BackpackItemView>(true))
        { rows.Add(ReadItemView("backpack", view.InstanceId, (RectTransform)view.transform, view.IsHovered, view.background)); }
        foreach (WarehouseItemView view in driver.warehouseUI.view.itemRoot.GetComponentsInChildren<WarehouseItemView>(true))
        { rows.Add(ReadItemView("warehouse", view.InstanceId, (RectTransform)view.transform, false, view.background)); }
        foreach (LootItemView view in driver.lootUI.view.contentRoot.GetComponentsInChildren<LootItemView>(true))
        { rows.Add(ReadItemView("loot", view.EntryId, (RectTransform)view.transform, false, view.background)); }
        return rows.ToArray();
    }

    private static ItemViewData ReadItemView(string kind, string id, RectTransform rect, bool hovered, Image background)
    {
        return new ItemViewData { kind = kind, id = id, objectId = rect.GetInstanceID(), active = rect.gameObject.activeInHierarchy,
            hovered = hovered, rect = RectData.From(ScreenRectangle(rect)), sourceAlpha = SourceAlpha(rect),
            color = background != null ? background.color : Color.clear,
            text = string.Join("\n", rect.GetComponentsInChildren<TMPro.TMP_Text>(true).Select(t => t.text).ToArray()) };
    }
